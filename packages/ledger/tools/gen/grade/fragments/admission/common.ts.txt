  const answered = answeredOf.bind(answeredKey);

  /**
   * The answered session's latest attempt before the call, whose outcomes and completion status are
   * the values in force, once its response processing ran.
   */
  const latestAtOf = query(delivery).rule((r) => {
    const { child, package: pkg, path, n, stage } = v(answered, "answered");
    const { at } = v(fields.Attempt, "attempt");

    return r
      .match(answered, { child, n, package: pkg, path, stage })
      .match(fields.Attempt, { at, child, n, package: pkg, path })
      .find({ child, latest: r.max(at), n, package: pkg, path, stage });
  });
  const latestAt = latestAtOf.bind(answeredKey);
  const latestOf = query(delivery).rule((r) => {
    const { child, package: pkg, path, n, stage, latest: at } = v(latestAt, "latest");

    return r
      .match(latestAt, { child, latest: at, n, package: pkg, path, stage })
      .find({ at, child, n, package: pkg, path, stage });
  });
  const boundLatest = latestOf.bind(answeredKey);

  /**
   * The answered session, by its key: the one item session the values in force are read over
   * (`valuesInForce`, each bound to it alone).
   */
  const latestSessionOf = query(delivery).rule((r) => {
    const { child, package: pkg, path, n } = v(boundLatest, "latestSession");

    return r
      .match(boundLatest, { child, n, package: pkg, path })
      .find({ child, n, package: pkg, path });
  });
  const latestSession = latestSessionOf.bind(answeredKey);
  const valuesInForce = valuesInForceOf(delivery);
  const processedOf = query(delivery).rule((r) => {
    const { child, package: pkg, path, n, stage, at } = v(boundLatest, "attempt");

    return r
      .match(boundLatest, { at, child, n, package: pkg, path, stage })
      .match(fields.ProcessedAttempt, { at, child, n, package: pkg, path })
      .find({ child, package: pkg, path, stage });
  });
  const processed = processedOf.bind(answeredKey);

  /**
   * The item body's element nodes, its `qti-item-body` among them, each kind by its name, as the
   * package's publication stores them (`ItemBodyElement`).
   */
  const body = query(delivery).rule((r) => {
    const { package: pkg, path } = v(answered, "answered");
    const { start, end, parent, kind } = v(fields.ItemBodyElement, "bodyNode");

    return r
      .match(answered, { package: pkg, path })
      .match(fields.ItemBodyElement, { end, kind, package: pkg, parent, path, start })
      .find({ end, kind, package: pkg, parent, path, start });
  });

  /**
   * The attributes of the body's elements admission reads, as the package's publication stores
   * them (`ItemBodyAttribute`).
   */
  const bodyAttributes = query(delivery).rule((r) => {
    const { package: pkg, path } = v(answered, "answered");
    const { start, name, value } = v(fields.ItemBodyAttribute, "attribute");

    return r
      .match(answered, { package: pkg, path })
      .match(fields.ItemBodyAttribute, { name, package: pkg, path, start, value })
      .find({ name, package: pkg, path, start, value });
  });
