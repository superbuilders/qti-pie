/** The taken item sessions: each of the call's sitting's, which the take's stratum before wrote (an item stage's
 * one session, the app's decideStage; a test's one per member, qti-pie's take). */
const takenOf = (session: GradedSession) =>
	query(Installed).rule((r) => {
		const { package: pkg, path, n } = v(fields.ItemSession, "taken")
		return r
			.match(fields.ItemSession, { child: session.child, package: pkg, path, n, stage: session.stage })
			.find({ child: session.child, package: pkg, path, n, stage: session.stage })
	})

/** The taken sessions' attempts, whose values gradeArgumentsOf's readers give a run, each with its session's stage, the
 * run key's: a session its take wrote in the stratum before holds none. */
const gradableOf = (session: GradedSession) =>
	query(Installed).rule((r) => {
		const taken = takenOf(session).bind(bySession(session))
		const { child, package: pkg, path, n, stage } = v(taken, "taken")
		const { at } = v(fields.Attempt, "attempt")
		return r
			.match(taken, { child, package: pkg, path, n, stage })
			.match(fields.Attempt, { child, package: pkg, path, n, at })
			.find({ child, package: pkg, path, n, stage, at })
	})
