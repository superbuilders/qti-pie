/** Each item session of the sitting with an attempt its response processing has not run for, and the earliest such
 * attempt's instant, which is the one graded: a session's attempts are processed in the order they were recorded. Each
 * with its sitting's stage: the session's run key (child, package, path, stage), which keys every value row of its run. */
const unprocessedOf = (session: GradedSession) =>
	query(Installed).rule((r) => {
		const { package: pkg, path, n } = v(fields.ItemSession, "session")
		const { at } = v(fields.Attempt, "attempt")
		return r
			.match(fields.ItemSession, { child: session.child, stage: session.stage, package: pkg, path, n })
			.match(fields.Attempt, { child: session.child, package: pkg, path, n, at })
			.where(r.not(fields.ProcessedAttempt, { child: session.child, package: pkg, path, n, at }))
			.find({ child: session.child, package: pkg, path, n, stage: session.stage, first: r.min(at) })
	})

/** The attempts the write grades: each session's earliest unprocessed one, once nothing it reads waits: no recorded
 * application of it awaits its evaluation (PendingApplication) and no rubric group of it whose every draw declined
 * awaits a person's value (AwaitingHumanEvaluation), each an input of its response processing (`#ledger/results.ts`:
 * grading runs "only once no recorded application of the attempt waits for its evaluation ... and no rubric group
 * whose every draw declined waits for a person's value"). A session whose earliest unprocessed attempt waits is graded
 * by no write until it stops waiting, so its attempts are processed in the order they were recorded. */
const gradableOf = (session: GradedSession) =>
	query(Installed).rule((r) => {
		const unprocessed = unprocessedOf(session).bind(bySession(session))
		const { child, package: pkg, path, n, stage, first: at } = v(unprocessed, "unprocessed")
		return r
			.match(unprocessed, { child, package: pkg, path, n, stage, first: at })
			.where(
				r.not(fields.PendingApplication, { child, package: pkg, path, n, at }),
				r.not(fields.AwaitingHumanEvaluation, { child, package: pkg, path, n, at })
			)
			.find({ child, package: pkg, path, n, stage, at })
	})

/** The subtree the item's response processing runs, by the document it is in (the delivery level's ResponseProgram,
 * read by the item's key): its own processing where it holds a rule (§4.6.1: "if both are given the internal rules are
 * still preferred"), else the published template's document its `template` names (the reference package's, §4.6.2:
 * "the template-location should be ignored"), else the template document its `template-location` locates (pantry's
 * LocatedTemplate). */
const responseRootsOf = (session: GradedSession) =>
	query(Installed).rule((r) => {
		const graded = gradableOf(session).bind(bySession(session))
		const { package: pkg, path: item } = v(graded, "graded")
		const { source, document } = v(fields.ResponseProgram, "program")
		const { start, end } = v(fields.Node, "root")
		return r
			.match(graded, { package: pkg, path: item })
			.match(fields.ResponseProgram, { package: pkg, path: item, source, document })
			.match(fields.Node, { package: source, path: document, start, end, kind: "qti-response-processing" })
			.find({ package: pkg, path: item, source, document, start, end })
	})

/** The graded items' declarations' nodes of the kinds the machine reads. */
const declarationNodesOf = (session: GradedSession) =>
	query(Installed).rule((r) => {
		const graded = gradableOf(session).bind(bySession(session))
		const { package: pkg, path: item } = v(graded, "graded")
		const { start, end, parent, kind } = v(fields.Node, "node")
		const { token } = v(fields.DeclarationNodeKind, "kind")
		return r
			.match(graded, { package: pkg, path: item })
			.match(fields.Node, { package: pkg, path: item, start, end, parent, kind })
			.match(fields.DeclarationNodeKind, { kind, token })
			.find({ package: pkg, path: item, start, end, parent, kind: token })
	})

/** The session's last attempt processed before this one, by its instant: whose completionStatus this one starts from
 * ("It remains with this value for the duration of the item session unless set to a different value", info §2.2.2.3). */
const previousOf = (session: GradedSession) =>
	query(Installed).rule((r) => {
		const attempt = gradableOf(session).bind(bySession(session))
		const { child, package: pkg, path, n, stage, at } = v(attempt, "attempt")
		const { at: earlier } = v(fields.ProcessedAttempt, "earlier")
		return r
			.match(attempt, { child, package: pkg, path, n, stage, at })
			.match(fields.ProcessedAttempt, { child, package: pkg, path, n, at: earlier })
			.where(r.lt(earlier, at))
			.find({ child, package: pkg, path, n, stage, previous: r.max(earlier) })
	})

/** The attempt's ordinal among its session's attempts, from 1: its numAttempts (§2.2.2.2: "the number of attempts
 * ... increases by 1 at the start of each attempt"). */
const ordinalsOf = (session: GradedSession) =>
	query(Installed).rule((r) => {
		const attempt = gradableOf(session).bind(bySession(session))
		const { child, package: pkg, path, n, stage, at } = v(attempt, "attempt")
		const { at: earlier } = v(fields.Attempt, "earlier")
		return r
			.match(attempt, { child, package: pkg, path, n, stage, at })
			.match(fields.Attempt, { child, package: pkg, path, n, at: earlier })
			.where(r.le(earlier, at))
			.find({ child, package: pkg, path, stage, attempts: r.count() })
	})

/** The run of the attempt: its key (child, package, path, stage), the seed its session draws under (SessionSeed, its
 * take's), the label of its ordinal, and whether it submitted every response at its default (`defaulted`, which the
 * machine reads to give no declared response a header). */
const runsOf = (session: GradedSession) =>
	query(Installed).rule((r) => {
		const attempt = gradableOf(session).bind(bySession(session))
		const { child, package: pkg, path, n, stage, at } = v(attempt, "attempt")
		const ordinal = ordinalsOf(session).bind(bySession(session))
		const { attempts } = v(ordinal, "ordinal")
		const { seed } = v(fields.SessionSeed, "seed")
		return r
			.match(attempt, { child, package: pkg, path, n, stage, at })
			.match(ordinal, { child, package: pkg, path, stage, attempts })
			.match(fields.SessionSeed, { child, stage, package: pkg, path, seed })
			.find({ child, package: pkg, path, stage, seed: Fn.call(seedStreamKey, { seed }), label: Fn.call(responseLabel, { attempts }), defaulted: Type.literal(DocumentRun.fields.defaulted, false) })
	})

/** The variables the attempt's run is given by its session's own record, each a header that may hold no value: the
 * previous attempt's completionStatus, and each value the session's template processing set, held as the rule that set
 * it says. The machine gives every declared response and numAttempts a header itself, and reads each variable's kind,
 * cardinality and base type from the run's declarations (`values_of`). */
const varsOf = (session: GradedSession) =>
	query(Installed)
		.rule((r) => {
			const previous = previousOf(session).bind(bySession(session))
			const { child, package: pkg, path, stage } = v(previous, "previous")
			return r
				.match(previous, { child, package: pkg, path, stage })
				.find({
					child,
					package: pkg,
					path,
					stage,
					holds: Type.literal(DocGiven.fields.holds, "value"),
					identifier: Type.literal(DocGiven.fields.identifier, "completionStatus")
				})
		})
		.rule((r) => {
			const attempt = gradableOf(session).bind(bySession(session))
			const { child, package: pkg, path, stage } = v(attempt, "attempt")
			const { variable } = v(fields.ItemSessionValue, "set")
			return r
				.match(attempt, { child, package: pkg, path, stage })
				.match(fields.ItemSessionValue, { child, stage, package: pkg, path, kind: "template", variable })
				.find({
					child,
					package: pkg,
					path,
					stage,
					holds: Type.literal(DocGiven.fields.holds, "value"),
					identifier: variable
				})
		})
		.rule((r) => {
			const attempt = gradableOf(session).bind(bySession(session))
			const { child, package: pkg, path, stage } = v(attempt, "attempt")
			const { variable } = v(fields.ItemSessionValue, "set")
			return r
				.match(attempt, { child, package: pkg, path, stage })
				.match(fields.ItemSessionValue, { child, stage, package: pkg, path, kind: "correct", variable })
				.find({
					child,
					package: pkg,
					path,
					stage,
					holds: Type.literal(DocGiven.fields.holds, "correct"),
					identifier: variable
				})
		})
		.rule((r) => {
			const attempt = gradableOf(session).bind(bySession(session))
			const { child, package: pkg, path, stage } = v(attempt, "attempt")
			const { variable } = v(fields.ItemSessionValue, "set")
			return r
				.match(attempt, { child, package: pkg, path, stage })
				.match(fields.ItemSessionValue, { child, stage, package: pkg, path, kind: "default", variable })
				.find({
					child,
					package: pkg,
					path,
					stage,
					holds: Type.literal(DocGiven.fields.holds, "default"),
					identifier: variable
				})
		})
