/** The attempt processed: its completion status after the run, a handle of QTI's four (§2.2.2.3), read by its token, one
 * query for all four, so the run is one value it reads. */
const statusesOf = (session: GradedSession) =>
	query(Learner).rule((r) => {
		const out = runOf(session)
		const attempt = gradableOf(session).bind(bySession(session))
		const { child, package: pkg, path, n, stage, at } = v(attempt, "attempt")
		const { value } = v(out.texts, "status")
		const { status } = v(fields.CompletionStatusToken, "statusToken")
		return r
			.match(attempt, { child, package: pkg, path, n, stage, at })
			.match(out.texts, {
				child,
				package: pkg,
				path,
				stage,
				holds: "value",
				identifier: "completionStatus",
				place: 0n,
				base: "identifier",
				value
			})
			.match(fields.CompletionStatusToken, { status, token: value })
			.find({ child, package: pkg, path, n, at, completionStatus: status })
	})

/** What the run demanded of the attempt (ApplicationDemand): each recorded application it reached with a response and
 * could not evaluate. A run that demands returns its demands alone, no value and no end, so none of the outcomes above
 * is written for it and the attempt waits for its evaluations; a later run, once they are recorded, grades it or
 * demands what only their values let the processing reach. */
const demandedOf = (session: GradedSession) =>
	query(Learner).rule((r) => {
		const out = runOf(session)
		const attempt = gradableOf(session).bind(bySession(session))
		const { child, package: pkg, path, n, stage, at } = v(attempt, "attempt")
		const { start } = v(out.demands, "demand")
		return r
			.match(attempt, { child, package: pkg, path, n, stage, at })
			.match(out.demands, { child, package: pkg, path, stage, start })
			.find({ child, package: pkg, path, n, at, operator: start })
	})

/** What the run asked a person for (HumanDemand): each rubric group of the attempt whose every application the run reached
 * declined, with no person's value. A run that asks returns its human demands alone, no value and no end, so none of the
 * outcomes above is written for it and the attempt waits for the person's value, which the run after it reads in place of
 * the group's draws. */
const humanDemandedOf = (session: GradedSession) =>
	query(Learner).rule((r) => {
		const out = runOf(session)
		const attempt = gradableOf(session).bind(bySession(session))
		const { child, package: pkg, path, n, stage, at } = v(attempt, "attempt")
		const { rubric } = v(out.human_demands, "humanDemand")
		return r
			.match(attempt, { child, package: pkg, path, n, stage, at })
			.match(out.human_demands, { child, package: pkg, path, stage, rubric })
			.find({ child, package: pkg, path, n, at, rubric })
	})

/** The maximum of each application the run demanded (ApplicationMaximum): its `maximum` operand as the run evaluated it,
 * the default in force, which the evaluator's request carries and a reply is bounded by. A NULL one has no row. */
const demandMaximaOf = (session: GradedSession) =>
	query(Learner).rule((r) => {
		const out = runOf(session)
		const attempt = gradableOf(session).bind(bySession(session))
		const { child, package: pkg, path, n, stage, at } = v(attempt, "attempt")
		const { start, maximum } = v(out.demand_maxima, "demandMaximum")
		return r
			.match(attempt, { child, package: pkg, path, n, stage, at })
			.match(out.demand_maxima, { child, package: pkg, path, stage, start, maximum })
			.find({ child, package: pkg, path, n, at, operator: start, maximum })
	})

/** The maximum a person's value of each group the run asked a person for may take (HumanDemandMaximum): the least of its
 * reached applications' maxima. A group whose every one is NULL has no row. */
const humanDemandMaximaOf = (session: GradedSession) =>
	query(Learner).rule((r) => {
		const out = runOf(session)
		const attempt = gradableOf(session).bind(bySession(session))
		const { child, package: pkg, path, n, stage, at } = v(attempt, "attempt")
		const { rubric, maximum } = v(out.human_demand_maxima, "humanDemandMaximum")
		return r
			.match(attempt, { child, package: pkg, path, n, stage, at })
			.match(out.human_demand_maxima, { child, package: pkg, path, stage, rubric, maximum })
			.find({ child, package: pkg, path, n, at, rubric, maximum })
	})

const processed = (session: GradedSession) =>
	ChangeSet.empty(Learner)
		.insert(mine.ApplicationDemand, demandedOf(session))
		.insert(mine.HumanDemand, humanDemandedOf(session))
		.insert(mine.ApplicationMaximum, demandMaximaOf(session))
		.insert(mine.HumanDemandMaximum, humanDemandMaximaOf(session))
		.insert(mine.ProcessedAttempt, statusesOf(session))

	return ChangeSet.union(
		ChangeSet.union(processed(gradedSitting), gradedSingles(gradedSitting)),
		ChangeSet.union(
			ChangeSet.union(gradedMultiples(gradedSitting), gradedOrdereds(gradedSitting)),
			gradedRecords(gradedSitting)
		)
	)
}
