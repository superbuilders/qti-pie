/** The attempts the read covers: each item session of the sitting at its latest attempt, processed or not (the Delivery level's LatestAttempt), with its sitting's stage, the run key's: the one its review shows. */
const gradableOf = (session: GradedSession) =>
	query(delivery).rule((r) => {
		const { package: pkg, path, n, at } = v(fields.LatestAttempt, "latest")
		return r
			.match(fields.LatestAttempt, { child: session.child, stage: session.stage, package: pkg, path, n, at })
			.find({ child: session.child, package: pkg, path, n, stage: session.stage, at })
	})
